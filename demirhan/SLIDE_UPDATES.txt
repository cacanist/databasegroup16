Q1 · Visitor Browser Matrix

SELECT
    v.Visitor_ID, v.VisitorType, cr.Region_ID, v.Country,
    bi.Browser_ID AS Browser, bi.BrowserVersion
FROM VISITOR v
JOIN  COUNTRY_REGION    cr ON  v.Country     = cr.Country
LEFT JOIN SESSION       s  ON  v.Visitor_ID  = s.Visitor_ID
LEFT JOIN SYSTEM_CONFIG sc ON  s.Config_ID   = sc.Config_ID
LEFT JOIN BROWSER_INFO  bi ON  sc.Browser_ID = bi.Browser_ID
ORDER BY v.Visitor_ID;

Q2 · Unique Monthly Visitors

SELECT
    tc.Month,
    mi.Quarter,
    COUNT(DISTINCT s.Visitor_ID) AS Visitor_Count
FROM TIME_CONTEXT tc
JOIN MONTH_INFO mi ON  tc.Month   = mi.Month
JOIN SESSION    s  ON  tc.Time_ID = s.Time_ID
GROUP BY tc.Month, mi.Quarter
ORDER BY Visitor_Count DESC;
